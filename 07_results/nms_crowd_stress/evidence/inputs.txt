## transcode (ffmpeg 7.1.5-0+deb13u1+rpt2, libx265, first 900 frames, 25 fps kept)
ffmpeg -i MOT20-02-raw.mp4 -frames:v 900 -vf "scale=1920:1080:flags=lanczos" -c:v libx265 -preset fast -crf 20 -pix_fmt yuv420p -tag:v hvc1 -x265-params log-level=error -an 04_data/videos/mot20_02_h265.mp4
ffmpeg -i MOT20-05-raw.mp4 -frames:v 900 -vf "scale=-2:1080:flags=lanczos,pad=1920:1080:(ow-iw)/2:0" -c:v libx265 -preset fast -crf 20 -pix_fmt yuv420p -tag:v hvc1 -x265-params log-level=error -an 04_data/videos/mot20_05_h265.mp4
# Why upscaling is harmless: the network input is a 640x360 letterbox of the 1920x1080 frame, which is smaller than the 960x540 preview.

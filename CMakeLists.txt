cmake_minimum_required(VERSION 3.25)
project(yolo_vulkan_optimize LANGUAGES CXX)

# Compiler cache: used automatically if present (silently skipped otherwise).
#   Because of HailoRT/Vulkan headers, one TU expands to 70k-130k lines after preprocessing, so compiling takes 2-5 s.
#   The verify skill requires 2 full builds per stage (--clean-first + a separate build-asan/), while
#   only 1-3 files have actually changed. ccache makes the rest take 0 s (measured 4.1 s -> 0.0 s).
#   Even with separate build directories (build-display/ etc.) the cache is shared, so parallel work does not double the cost.
#   Also set as the C launcher (C stubs generated by wayland-scanner). Disable with -DYVO_USE_CCACHE=OFF.
option(YVO_USE_CCACHE "Use ccache as the compiler launcher if available" ON)
if(YVO_USE_CCACHE)
  find_program(YVO_CCACHE ccache)
  if(YVO_CCACHE)
    set(CMAKE_CXX_COMPILER_LAUNCHER ${YVO_CCACHE})
    set(CMAKE_C_COMPILER_LAUNCHER ${YVO_CCACHE})
    message(STATUS "ccache enabled: ${YVO_CCACHE}")
  else()
    message(STATUS "ccache not found — building without a cache")
  endif()
endif()

# Language standard and tool settings
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
# All executables are collected in <build>/bin (2026-09-22). Moving source folders does not break the binary paths
# (build/bin/yvo_*).
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)

# Release if no build type is given (single-config generators only)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
  set_property(CACHE CMAKE_BUILD_TYPE PROPERTY STRINGS Debug Release RelWithDebInfo MinSizeRel)
endif()

# Required dependencies
find_package(HailoRT REQUIRED)   # HailoRT::libhailort
find_package(Vulkan REQUIRED)    # Vulkan::Vulkan

# Shared targets and helpers: yvo_warnings, YVO_CPU_FLAGS, yvo_target_defaults(), yvo_assets
include(${PROJECT_SOURCE_DIR}/cmake/YvoTargets.cmake)
# Shader compile helper: yvo_add_shaders(<target> <shader>...)
include(${PROJECT_SOURCE_DIR}/cmake/YvoShaders.cmake)

# Subdirectories. They are required folders, so they are added unconditionally — configure stops if a folder is missing
# (the old yvo_add_subdirectory_if_exists silently skipped missing folders).
add_subdirectory(01_src)
add_subdirectory(02_shaders)
